local function createAfterimage(char)
	if not Config.Afterimage or not char then return end
	task.spawn(function()
		for _, part in ipairs(char:GetChildren()) do
			if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" and part.Transparency < 1 then
				local ghost = Instance.new("Part")
				ghost.Size = part.Size
				ghost.CFrame = part.CFrame
				ghost.Anchored = true
				ghost.CanCollide = false
				ghost.CanQuery = false
				ghost.Material = Enum.Material.Neon
				ghost.Color = Color3.fromRGB(0, 225, 255)
				ghost.Transparency = 0.45
				ghost.Parent = Workspace
				
				local tw = TweenService:Create(ghost, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
					Transparency = 1,
					Size = part.Size * 0.85,
					Color = Color3.fromRGB(170, 70, 255)
				})
				tw:Play()
				task.delay(0.36, function()
					if ghost then ghost:Destroy() end
				end)
			end
		end
	end)
end

local function makeDraggable(dragHandle, targetFrame)
	local dragging, dragInput, dragStart, startPos
	dragHandle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = targetFrame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)
	dragHandle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	track(UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			local goalPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
			TweenService:Create(targetFrame, TweenInfo.new(0.08, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
				Position = goalPos
			}):Play()
		end
	end))
end

	fill.Size = UDim2.new(initPercent, 0, 1, 0)
	fill.BackgroundColor3 = Color3.fromRGB(255, 255, 255)

	fillGrad.Color = ColorSequence.new(Theme.Accent, Theme.AccentAlt)
	fillGrad.Parent = fill

	local knob = Instance.new("Frame")
	knob.AnchorPoint = Vector2.new(0.5, 0.5)
	knob.Position = UDim2.new(initPercent, 0, 0.5, 0)
	knob.Size = UDim2.fromOffset(14, 14)
	knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	knob.BorderSizePixel = 0
	knob.Parent = barBg
	addCorner(knob, 999)

	local hitBox = Instance.new("TextButton")
	hitBox.Size = UDim2.new(1, 0, 0, 26)
	hitBox.Position = UDim2.fromOffset(0, 28)
	hitBox.BackgroundTransparency = 1
	hitBox.Text = ""
	hitBox.Parent = card

	local sliding = false
	local function updateSlider(inputX)
		local pct = math.clamp((inputX - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
		local rawVal = minVal + (maxVal - minVal) * pct
		local rounded = math.floor(rawVal * 10 + 0.5) / 10
		
		valLbl.Text = tostring(rounded) .. suffix
		fill.Size = UDim2.new(pct, 0, 1, 0)
		knob.Position = UDim2.new(pct, 0, 0.5, 0)
		callback(rounded)
	end

	hitBox.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true
			updateSlider(input.Position.X)
		end
	end)
	track(UserInputService.InputChanged:Connect(function(input)
		if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			updateSlider(input.Position.X)
		end
	end))
	track(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = false
		end
	end))
end

local function performRandomDodge(attackerModel)
	if not Config.Enabled or State.IsDodging then return end
	local myChar = LocalPlayer.Character
	local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
	local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
	if not (myHRP and myHum and myHum.Health > 0) then return end

	if not attackerModel or not attackerModel:IsA("Model") or attackerModel == myChar then return end
	local enemyHRP = attackerModel:FindFirstChild("HumanoidRootPart") or attackerModel:FindFirstChild("Head")
	local enemyHum = attackerModel:FindFirstChildOfClass("Humanoid")
	if not (enemyHRP and enemyHum and enemyHum.Health > 0) then return end

	local dist = (myHRP.Position - enemyHRP.Position).Magnitude
	if dist > Config.Radius then return end

	State.IsDodging = true
	State.DodgeCount += 1
	SubTitle.Text = ("Đã né thành công: %d đòn"):format(State.DodgeCount)

	createAfterimage(myChar)

	local ePos = enemyHRP.Position
	local flatLook = enemyHRP.CFrame.LookVector
	local flatDir = Vector3.new(flatLook.X, 0, flatLook.Z)
	if flatDir.Magnitude < 0.01 then flatDir = Vector3.new(0, 0, -1) else flatDir = flatDir.Unit end
	local flatEnemyCF = CFrame.lookAt(ePos, ePos + flatDir)

	local choice = 1
	if Config.Mode == "Ngẫu nhiên" then choice = math.random(1, 3)
	elseif Config.Mode == "Sau lưng" then choice = 1
	elseif Config.Mode == "Trái / Phải" then choice = (math.random(1, 2) == 1) and 2 or 3 end

	local offsetVec
	if choice == 1 then offsetVec = Vector3.new(0, 0, Config.Offset)
	elseif choice == 2 then offsetVec = Vector3.new(Config.Offset, 0, 0)
	else offsetVec = Vector3.new(-Config.Offset, 0, 0) end

	local targetPos = (flatEnemyCF * CFrame.new(offsetVec)).Position
	local safePos = Vector3.new(targetPos.X, ePos.Y, targetPos.Z)
	local finalCFrame = CFrame.lookAt(safePos, Vector3.new(ePos.X, safePos.Y, ePos.Z))

	myHRP.AssemblyLinearVelocity = Vector3.zero
	TweenService:Create(myHRP, TweenInfo.new(Config.TweenSpeed, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out), { CFrame = finalCFrame }):Play()

	task.delay(Config.Cooldown, function()
		State.IsDodging = false
	end)
end

local function hookRemote(remote)
	if not remote:IsA("RemoteEvent") then return end
	if hookedRemotes[remote] then return end
	hookedRemotes[remote] = true

	track(remote.OnClientEvent:Connect(function(...)
		if not Config.Enabled or State.IsDodging then return end
		local myChar = LocalPlayer.Character
		if not myChar then return end
		for _, arg in ipairs({...}) do
			if typeof(arg) == "Instance" then
				if arg:IsA("Model") and arg ~= myChar then
					performRandomDodge(arg)
					return
				elseif arg:IsA("BasePart") and arg.Parent and arg.Parent:IsA("Model") and arg.Parent ~= myChar then
					performRandomDodge(arg.Parent)
					return
				end
			end
		end
	end))
end

task.spawn(function()
	local Knit = ReplicatedStorage:WaitForChild("Knit", 8)
	local targetFolder = Knit or ReplicatedStorage
	for _, desc in ipairs(targetFolder:GetDescendants()) do hookRemote(desc) end
	track(targetFolder.DescendantAdded:Connect(hookRemote))
end)

env.AutoDodge_Cleanup = function()
	Config.Enabled = false
	for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
	table.clear(connections)
	table.clear(hookedRemotes)
	for _, obj in ipairs(visualObjects) do pcall(function() obj:Destroy() end) end
	if ScreenGui then pcall(function() ScreenGui:Destroy() end) end
	env.AutoDodge_Cleanup = nil
end
