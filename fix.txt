--// ================= LOGIC & RANDOM DODGE SYSTEM =================

-- ตรวจจับระยะทางแบบวงกลม 360 องศา (ไม่ต้องหันหน้ามอง)
local function isInCircleRadius(myHRP, enemyHRP)
    local distance = (myHRP.Position - enemyHRP.Position).Magnitude
    if distance <= DODGE_RADIUS then
        return true
    end
    return false
end

-- เอฟเฟกต์จำลองการพุ่ง
local function playProceduralDodgeEffect(myHRP)
    local originalCFrame = myHRP.CFrame
    local tiltTween = TweenService:Create(myHRP, TweenInfo.new(0.07, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        CFrame = originalCFrame * CFrame.Angles(math.rad(-10), 0, 0)
    })
    tiltTween:Play()
end

-- ฟังก์ชันสุ่มทิศทางการหลบ (หลัง, ขวา, ซ้าย) และหันหน้ามองกลับมาที่คนตี
local function performRandomDodge(attackerModel)
    if not AutoDodgeEnabled or IsDodging then return end
    if not Character or not Character:FindFirstChild("HumanoidRootPart") then return end
    if not attackerModel or not attackerModel:IsA("Model") or attackerModel == Character then return end
    
    local enemyHRP = attackerModel:FindFirstChild("HumanoidRootPart") or attackerModel:FindFirstChild("Head")
    local myHRP = Character.HumanoidRootPart
    
    if not enemyHRP then return end

    -- เช็คระยะวงกลมรอบตัว
    if not isInCircleRadius(myHRP, enemyHRP) then return end

    IsDodging = true
    playProceduralDodgeEffect(myHRP)

    local enemyCFrame = enemyHRP.CFrame
    local targetPosition = enemyCFrame.Position
    local finalCFrame

    -- สุ่มเลือกทิศทาง (1 = ด้านหลัง, 2 = ด้านขวา, 3 = ด้านซ้าย)
    local randomChoice = math.random(1, 3)

    if randomChoice == 1 then
        -- 1. วาร์ปไปด้านหลัง
        local offsetCFrame = enemyCFrame * CFrame.new(0, 0, OFFSET_DISTANCE)
        finalCFrame = CFrame.new(offsetCFrame.Position, targetPosition)
    elseif randomChoice == 2 then
        -- 2. วาร์ปออกข้างขวา
        local offsetCFrame = enemyCFrame * CFrame.new(OFFSET_DISTANCE, 0, 0)
        finalCFrame = CFrame.new(offsetCFrame.Position, targetPosition)
    else
        -- 3. วาร์ปออกข้างซ้าย
        local offsetCFrame = enemyCFrame * CFrame.new(-OFFSET_DISTANCE, 0, 0)
        finalCFrame = CFrame.new(offsetCFrame.Position, targetPosition)
    end

    -- พุ่งด้วย TweenService สมูททีเดียวจบ
    local tweenInfo = TweenInfo.new(0.12, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out)
    local dodgeTween = TweenService:Create(myHRP, tweenInfo, {CFrame = finalCFrame})
    
    dodgeTween:Play()

    -- ปลดล็อคคูลดาวน์
    task.delay(COOLDOWN_TIME, function()
        IsDodging = false
    end)
end

-- ดักจับ Knit Remotes ทั้งหมด
local function hookRemote(remote)
    if remote:IsA("RemoteEvent") then
        local connectionKey = "RandomDodge_" .. remote.Name
        if not remote:GetAttribute(connectionKey) then
            remote:SetAttribute(connectionKey, true)
            
            remote.OnClientEvent:Connect(function(...)
                if not AutoDodgeEnabled or IsDodging then return end
                
                local args = {...}
                for _, arg in pairs(args) do
                    if typeof(arg) == "Instance" then
                        if arg:IsA("Model") and arg ~= Character then
                            performRandomDodge(arg)
                            return
                        elseif arg:IsA("BasePart") and arg.Parent and arg.Parent:IsA("Model") then
                            performRandomDodge(arg.Parent)
                            return
                        end
                    end
                end
            end)
        end
    end
end

-- สแกนหารีโมทใน Knit
task.spawn(function()
    local Knit = ReplicatedStorage:WaitForChild("Knit", 15)
    if Knit then
        for _, desc in ipairs(Knit:GetDescendants()) do
            hookRemote(desc)
        end
        Knit.DescendantAdded:Connect(function(desc)
            hookRemote(desc)
        end)
    else
        for _, desc in ipairs(ReplicatedStorage:GetDescendants()) do
            hookRemote(desc)
        end
        ReplicatedStorage.DescendantAdded:Connect(function(desc)
            hookRemote(desc)
        end)
    end
end)
